# Test-only dependencies.
#
# The prototypes in experiments/ are standard library ONLY and must stay that way
# (AGENTS.md) — a musician with a stock Mac has to be able to run them with no
# install step. That constraint applies to experiments/, not to tests/.
#
#   python3 -m pip install -r tests/requirements-dev.txt
#   pytest tests/
#
# CI runs a deliberate Python 3.9 leg (see .github/workflows/ci.yml), and pytest 9.x
# requires Python >=3.10 — a single unconditional `pytest>=9` floor breaks that leg
# (see PR #9). Split by environment marker instead of dropping 3.9.

pytest>=8.4,<9; python_version < "3.10"
pytest>=9.1.1; python_version >= "3.10"
pytest-cov>=7.1.0
